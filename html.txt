  <script>
    const pages = {
      home: {
        eyebrow: "Welcome to the chill zone",
        title: "Explore Arctic",
        description: "Choose a page above to get started."
      },
      tiktok: {
        eyebrow: "Short videos",
        title: "TikTok",
        description: "Open TikTok’s official website in a new tab.",
        url: "https://www.tiktok.com/"
      },
      youtube: {
        eyebrow: "Videos and creators",
        title: "YouTube",
        description: "Open YouTube’s official website in a new tab.",
        url: "https://www.youtube.com/"
      },
      fortnite: {
        eyebrow: "Games and updates",
        title: "Fortnite",
        description: "Open Fortnite’s official website in a new tab.",
        url: "https://www.fortnite.com/"
      }
    };

    const eyebrow = document.querySelector("#eyebrow");
    const title = document.querySelector("#title");
    const description = document.querySelector("#description");
    const visit = document.querySelector("#visit");

    document.querySelectorAll("[data-page]").forEach(button => {
      button.addEventListener("click", () => {
        const page = pages[button.dataset.page];
        eyebrow.textContent = page.eyebrow;
        title.textContent = page.title;
        description.textContent = page.description;

        document.querySelectorAll("[data-page]").forEach(item =>
          item.removeAttribute("aria-current")
        );
        button.setAttribute("aria-current", "page");

        visit.hidden = !page.url;
        if (page.url) visit.href = page.url;
      });
    });
  </script>
